    <script>
        const audio = document.getElementById('bgAudio');
        const musicIcon = document.getElementById('musicIcon');
        const musicText = document.getElementById('musicText');

        // Play audio on first user touch/click anywhere
        function autoPlayAudio() {
            audio.play().then(() => {
                musicIcon.className = 'fa-solid fa-pause';
                musicText.innerText = 'PAUSE';
                document.getElementById('musicToggle').style.borderColor = 'var(--neon-blue)';
                document.getElementById('musicToggle').style.boxShadow = '0 0 20px var(--neon-blue)';
                window.removeEventListener('click', autoPlayAudio);
                window.removeEventListener('touchstart', autoPlayAudio);
            }).catch(e => console.log("Audio play blocked until click"));
        }

        window.addEventListener('click', autoPlayAudio);
        window.addEventListener('touchstart', autoPlayAudio);

        // Preloader Logic
        window.addEventListener('load', () => {
            const preloader = document.getElementById('preloader');
            setTimeout(() => {
                preloader.style.opacity = '0';
                preloader.style.visibility = 'hidden';
            }, 2500);
        });

        // Toggle Audio Button
        function toggleAudio() {
            if (audio.paused) {
                audio.play();
                musicIcon.className = 'fa-solid fa-pause';
                musicText.innerText = 'PAUSE';
                document.getElementById('musicToggle').style.borderColor = 'var(--neon-blue)';
                document.getElementById('musicToggle').style.boxShadow = '0 0 20px var(--neon-blue)';
            } else {
                audio.pause();
                musicIcon.className = 'fa-solid fa-play';
                musicText.innerText = 'HIDDEN 80s';
                document.getElementById('musicToggle').style.borderColor = 'var(--neon-pink)';
                document.getElementById('musicToggle').style.boxShadow = '0 0 15px var(--neon-pink)';
            }
        }
    </script>
